макросы дают способ создавать обобщённые списки, которые хранят данные различных типов. в зависимости от потребностей и задач определяют различные формы списка. ниже разбирается более простой пример. сначала только определение списка:

```c


#define DEFINE_LIST(T) \

typedef struct { \

    T* data; \

    size_t count; \

    size_t size; \

} list_##T;
```

данный макрос принимает параметр T, через который передаётся имя типа. макрос определяет структуру, которая представляет простейший список и имеет три поля. первое поле, data, хранит собственно данные списка. в качестве типа этого поля используется значение параметра T. поле count указывает на количество элементов, а поле size: на вместительность, то есть для скольких элементов выделена память в списке. это классический динамический массив: логический размер и ёмкость разделены, чтобы амортизировать добавление.

название структуры зависит от названия типа и образуется простой конкатенацией: `list_##T`. оператор `##` препроцессора склеивает токены, поэтому для каждого T получается отдельный тип и не возникает коллизии имён.

возьмём такой вызов макроса

```c


DEFINE_LIST(int)
```

он будет разворачиваться в следующий код:

```c


typedef struct {

    int* data;

    size_t count;

    size_t size;

} list_int;
```

с помощью этого макроса создают отдельные типы списков для отдельных типов данных без необходимости определять для каждого типа списка отдельную структуру. по сути это ручная инстанциация «шаблона» на этапе препроцесса, аналог template в c++ с тем отличием, что проверка типов происходит уже после подстановки.

далее к списку добавляется ряд самых базовых операций:

```c


#define DEFINE_LIST(T) \

typedef struct { \

    T* data; \

    size_t count; \

    size_t size; \

} list_##T; \

\

void list_##T##_new(list_##T* v) { \

    v->data = NULL; \

    v->count= v->size = 0; \

} \

\

void list_##T##_push(list_##T* v, T elem) { \

    if (v->count == v->size) { \

        v->size = v->size ? v->size << 1 : 1; \

        v->data = (T*)realloc(v->data, v->size * sizeof(T)); \

    } \

    v->data[v->count++] = elem; \

} \

\

T list_##T##_pop(list_##T* v) { \

    return v->data[--v->count]; \

} \

\

void list_##T##_free(list_##T* v) { \

    free(v->data); \

    list_##T##_new(v); \

}
```

здесь добавлен ряд функций. прежде всего функция для инициализации списка:

```c


void list_##T##_new(list_##T* v) { \

    v->data = NULL; \

    v->count= v->size = 0; \

}
```

название этой функции опять же зависит от типа данных, поэтому конфликтов между реализациями этой функции для разных типов не будет. в качестве параметра функция принимает указатель на список и инициализирует его поля. нулевой data согласован с `realloc(NULL, n)`, который ведёт себя как `malloc`.

функция `list_##T##_push()` добавляет в список один элемент:

```c


void list_##T##_push(list_##T* v, T elem) { \

    if (v->count == v->size) { \

        v->size = v->size ? v->size << 1 : 1; \

        v->data = (T*)realloc(v->data, v->size * sizeof(T)); \

    } \

    v->data[v->count++] = elem; \

}
```

если места в списке нет, библиотечной функцией realloc() выделяется дополнительная память под удвоенное количество элементов, и элемент добавляется в массив data с увеличением счётчика count. сдвиг `size << 1` даёт геометрический рост ёмкости; амортизированная сложность push: O(1). результат realloc в примере не проверяется: при нехватке памяти указатель станет нулевым, и последующая запись будет неопределённым поведением.

функция `list_##T##_pop`, наоборот, извлекает последний элемент, уменьшая счётчик элементов:

```c


T list_##T##_pop(list_##T* v) { \

    return v->data[--v->count]; \

}
```

pop на пустом списке уменьшает count ниже нуля (для `size_t` это обёртка к огромному числу) и читает за границей буфера. инвариант вызывающего кода: pop только при `count > 0`. память при pop не сужается.

функция очистки освобождает ранее выделенную память и сбрасывает значение полей структуры:

```c


void list_##T##_free(list_##T* v) { \

    free(v->data); \

    list_##T##_new(v); \

}
```

пример применения:

```c


#include <stdio.h>

#include <stdlib.h>



#define DEFINE_LIST(T) \

typedef struct { \

    T* data; \

    size_t count; \

    size_t size; \

} list_##T; \

\

void list_##T##_new(list_##T* v) { \

    v->data = NULL; \

    v->count= v->size = 0; \

} \

\

void list_##T##_push(list_##T* v, T elem) { \

    if (v->count == v->size) { \

        v->size = v->size ? v->size << 1 : 1; \

        v->data = (T*)realloc(v->data, v->size * sizeof(T)); \

    } \

    v->data[v->count++] = elem; \

} \

\

T list_##T##_pop(list_##T* v) { \

    return v->data[--v->count]; \

} \

\

void list_##T##_free(list_##T* v) { \

    free(v->data); \

    list_##T##_new(v); \

}



DEFINE_LIST(int) // определяем список





int main(){

    list_int numbers; // определяем переменную списка

    list_int_new(&numbers);     // инициализируем его поля

    list_int_push(&numbers, 22);    // добавляем в список новый элемент

    int n = list_int_pop(&numbers);  // удаляем из списка последний элемент

    printf("n: %d\n", n);



    // аналогично можно добавить несколько элементов

    list_int_push(&numbers, 11);

    list_int_push(&numbers, 12);

    list_int_push(&numbers, 13);

    // перебирем элементы

    for(size_t i=0; i<numbers.count; ++i){

        printf("numbers[%lu]: %d\n", i,  numbers.data[i]);

    }



    list_int_free(&numbers);  // очищаем память под список

    return 0;

}
```

поскольку в макросе задано создание структуры с именем "list_##T", при использовании типа `int` будет создаваться структура "list_int". соответственно этот тип используют для определения переменных или параметров. здесь определяется переменная numbers типа list_int, и с ней выполняются различные операции.

аналогично этот макрос используют для создания списка под другие типы данных:

```c


// определение макроса то же самое



typedef struct{

    char* name;

    int age;

} person;



DEFINE_LIST(int)

DEFINE_LIST(person)



int main(){

    list_person people; // определяем переменную списка

    list_person_new(&people); 



    // добавляем данные

    list_person_push(&people, (person) {"Tom", 40});  

    list_person_push(&people, (person) {"Bob", 46});  

    list_person_push(&people, (person) {"Sam", 29});    



    for(size_t i=0; i<people.count; ++i){

        printf("%s: %d\n", people.data[i].name, people.data[i].age);

    }



    list_person_free(&people);  // очищаем память под список

    return 0;

}
```

compound literal `(person) {"Tom", 40}` создаёт временный объект, который копируется в буфер списка по значению. указатель `name` при этом копируется как адрес строкового литерала, отдельного дублирования строки нет.

другой аналогичный пример создания списков через указатели:

```c


#include <stdio.h>

#include <stdlib.h>





#define DEFINE_LIST(T) \

typedef struct { \

    T* data; \

    size_t count; \

    size_t size; \

} T##List; \

\

T##List* T##List_new() { \

    T##List* list = malloc(sizeof(*list)); \

    list->data = 0; \

    list->count= list->size = 0; \

    return list; \

} \

\

void T##List_append(T##List* list, T elem) { \

    if (list->count == list->size) { \

        list->size = list->size ? list->size << 1 : 1; \

        list->data = (T*)realloc(list->data, list->size * sizeof(T)); \

    } \

    list->data[list->count++] = elem; \

} \

void T##List_delete(T##List* list) { \

    if(list->data) { \

        free(list->data); \

        list->data = 0; \

    } \

    list->count= list->size = 0; \

    free(list); \

}





typedef struct{

    char* name;

    int age;

} Person;



DEFINE_LIST(Person)



int main(){

    PersonList* people = PersonList_new(); 



    PersonList_append(people, (Person) {"Tom", 40});  

    PersonList_append(people, (Person) {"Bob", 46});  

    PersonList_append(people, (Person) {"Sam", 29});    



    for(size_t i=0; i<people->count; ++i){

        printf("%s: %d\n", people->data[i].name, people->data[i].age);

    }



    PersonList_delete(people);  // очищаем память под список

    return 0;

}
```

здесь аналогичный пример, только функция "T##List_new" возвращает указатель на структуру списка, через который затем взаимодействуют с его элементами. схема владения другая: контейнер живёт в куче, а `T##List_delete` освобождает и буфер, и сам дескриптор. для типа `Person` имена получаются `PersonList`, `PersonList_new`, `PersonList_append`, `PersonList_delete`.
