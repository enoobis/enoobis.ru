Кроме типа `char` для работы с символами в Си есть тип **wchar_t** (или "широкий символ"/ "wide character"), который определен в заголовочном файле **stddef.h**. Для работы с ним предназначены символьные литералы, которые предваряются префиксом **L**. Например:

```c
#include <stdio.h>
#include <stddef.h>
 
int main(void)
{
    wchar_t c = L'A';
    printf("%c - %d \n", c, c); // A - 65
    return 0;
}
```

Здесь константа `L'A'` имеет тип wchar_t. И с этим символом также связан некоторый числовой код.

Стоит отметить, что поддержка этого типа определяется реализацией. На Linux широкий символ имеет длину 32 бита, что эквивалентно беззнаковому целому числу типа `unsigned int`, и соответственно имеет достаточно места для размещения кодовой точки Unicode. Но в Windows исторически широкий символ имеет длину 16 бит, а не 32, поэтому для этой системы `wchar_t` эквивалентен `unsigned short int`.

Аналогичным образом можно определять строки из широких символов. Для их вывода на консоль применяется специафикатор %ls :

```c
#include <stdio.h>
#include <stddef.h>

int main(void)
{
    wchar_t str[] = L"Hello";
    printf("str: %ls\n", str);
    return 0;
}
```

### Поддержка для UTF-8, UTF-16 и UTF-32

Стандарт C11 добавил поддержку символов в кодировках UTF-16 и UTF-32. Начиная со стандарта C23 также добавлена поддержка для кодировки UTF-8. Для этого в заголовочном файле **uchar.h** были определены соответственно типы **char8_t**, **char16_t** и **char32_t**, которые занимают 1, 2 и 4 байта соответственно.

При определении символов типа `char16_t` перед символьным литералом указывается префикс **u**, для символов типа `char32_t*` - префикс **U**, а для **char8_t** - префикс **u8**:

```c
#include <stdio.h>
//#include <stddef.h>
#include <uchar.h>
 
int main(void)
{
    char8_t letter8 = u8'N';
    char16_t letter16 = u'Ы';
    char32_t letter32 = U'Ю';
    printf("%c - %d\n", letter8, letter8);
    printf("%c - %d\n", letter16, letter16);
    printf("%c - %d\n", letter32, letter32);
    return 0;
}
```

Также можно определять и строки символов этих типов:

```c
#include <stdio.h>
#include <uchar.h>

int main(void)
{
	char16_t utf16_str[] = u"UTF-16 string";
	
	printf("%ls \n", utf16_str);

 	char32_t utf32_str[] = U"UTF-32 string";
	// определяем размер строки utf-32
	int n = sizeof(utf32_str)/4;
	for(int i=0; i < n; i++)
	{
		printf("%lc", utf32_str[i]);
	}

	return 0;
}
```

Для вывода строки типа `char16_t` применяется спецификатор `%ls`. А для вывода отдельного символа - спецификатор `%lc.`. Но опять же поддержка этих типов определяется конкретной реализацией.

При этом можно использовать числовые коды символов в соответствующей кодировке. Например:

```c
#include <stdio.h>
#include <uchar.h>
#include <locale.h>     // для локализации на Windows
 
int main(void)
{
    setlocale(LC_ALL, "");  // для локализации на Windows
    char16_t utf16_str[] = u"\u043F\u0440\u0438\u0432\u0435\u0442"; //u"привет";
     
    printf("%ls \n", utf16_str);    // привет

    char16_t s = u'\u044F';
    printf("%lc\n", s);            // я

    return 0;
}
```
