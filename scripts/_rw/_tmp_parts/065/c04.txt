```cpp


#include <iostream>

#include <string>



class Person

{

public:

    Person(std::string name, unsigned age): name{name}, age{age}

    { }

    void print()

    {

        std::cout << "Name: " << name << "\tAge: " << age << std::endl;

    }

private:

    std::string name;

    unsigned age;

};

 

int main()

{

    Person bob{"Bob", 42};

    bob.print();

}
```
