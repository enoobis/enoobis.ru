#include <iostream>

#include <string>



struct person

{

public:

    person(std::string name, unsigned age): name{name}, age{age}

    { 

    }

    void print()

    {

        std::cout << "Name: " << name << "\tAge: " << age << std::endl;

    }

private:

    std::string name;

    unsigned age;

};

 

int main()

{

    person bob{"Bob", 42};

    bob.print();

}
