#include <iostream>

#include <string>

 

class Person 

{

private:

    std::string name;

    unsigned age;

public:

    Person(std::string p_name, unsigned p_age)

    {

        name = p_name;

        age = p_age;

    }

    void print() 

    {

        std::cout << "Name: " << name << "\tAge: " << age << std::endl;

    }

};

int main()

{

    const Person tom{"Tom", 38};

    tom.print();    // ! Ошибка

}
