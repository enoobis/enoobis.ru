```cpp


#include <iostream>

#include <string>



class Person 

{

private:

    std::string name;

    unsigned age;

public:

    Person(std::string p_name, unsigned p_age);

    void print();

};

// конструктор

Person::Person(std::string p_name, unsigned p_age)

{ 

    name = p_name; 

    age = p_age;

}

void Person::print() 

{

    std::cout << "Name: " << name << "\tAge: " << age << std::endl;

}

int main()

{

    Person tom{"Tom", 38};

    tom.print();    // Name: Tom   Age: 22

}
```
